% =====================================================================
% Allocation of the d_k/2 rotation planes of one attention head.
% Top: trunk/mode --- all planes by TOA with scale gamma_t.
% Bottom: deinterleaving --- even split u^x / u^y (three cells each,
% first / dots / last) with scales gamma_{u^x}, gamma_{u^y}.
% Cell labels are single-line (no shortstack) so Omega(...) does not wrap.
% Loaded from sections/04_method/04_attention_variants.tex via \input.
% =====================================================================
\begin{tikzpicture}[
    font=\small,
    plane/.style={draw=black!55, thin, minimum width=18mm, minimum height=7.5mm,
                  inner sep=0.5pt, outer sep=0pt, align=center},
    toaplane/.style={plane, fill=blue!18},
    uxplane/.style={plane, fill=red!18},
    uyplane/.style={plane, fill=orange!28},
    rowlabel/.style={font=\footnotesize, anchor=east, align=right},
    slicebrace/.style={decorate, decoration={brace, mirror, amplitude=4pt}, thick},
    bracelabel/.style={font=\footnotesize, anchor=north},
  ]

  \def\W{18mm}
  \def\NP{6}

  % =================== TOP STRIP: trunk and mode branch =================
  \node[rowlabel] at (-2mm, 0) {trunk and\\mode branch};
  \foreach \m in {1,...,\NP} {
    \node[toaplane] (t\m) at ({(\m-0.5)*\W}, 0) {};
  }
  \node[font=\scriptsize] at (t1) {$\Omega(\gamma_t t_n\theta_1)$};
  \node[font=\scriptsize] at (t2) {$\cdots$};
  \node[font=\scriptsize] at (t3) {$\cdots$};
  \node[font=\scriptsize] at (t4) {$\cdots$};
  \node[font=\scriptsize] at (t5) {$\cdots$};
  \node[font=\scriptsize] at (t6) {$\Omega(\gamma_t t_n\theta_{d_k/2})$};

  \draw[slicebrace] (0, -4.5mm) -- ({\NP*\W}, -4.5mm);
  \node[bracelabel] at ({\NP*\W/2}, -7mm)
    {all $d_k/2$ planes: coordinate $t_n$, scale $\gamma_t$};

  % ============== BOTTOM STRIP: deinterleaving branch ====================
  \begin{scope}[yshift=-26mm]
    \node[rowlabel] at (-2mm, 0) {deinterleaving\\branch};
    \foreach \m in {1,...,3} {
      \node[uxplane] (b\m) at ({(\m-0.5)*\W}, 0) {};
    }
    \foreach \m in {4,...,6} {
      \node[uyplane] (b\m) at ({(\m-0.5)*\W}, 0) {};
    }
    \node[font=\scriptsize] at (b1) {$\Omega(\gamma_{u^x}u^{x}_n\theta_1)$};
    \node[font=\scriptsize] at (b2) {$\cdots$};
    \node[font=\scriptsize] at (b3) {$\Omega(\gamma_{u^x}u^{x}_n\theta_{d_k/4})$};
    \node[font=\scriptsize] at (b4) {$\Omega(\gamma_{u^y}u^{y}_n\theta_1)$};
    \node[font=\scriptsize] at (b5) {$\cdots$};
    \node[font=\scriptsize] at (b6) {$\Omega(\gamma_{u^y}u^{y}_n\theta_{d_k/4})$};

    \draw[slicebrace] (0, -4.5mm) -- ({3*\W}, -4.5mm);
    \node[bracelabel] at ({1.5*\W}, -7mm)
      {$d_k/4$ planes: $u^{x}_n$, scale $\gamma_{u^x}$};

    \draw[slicebrace] ({3*\W}, -4.5mm) -- ({6*\W}, -4.5mm);
    \node[bracelabel] at ({4.5*\W}, -7mm)
      {$d_k/4$ planes: $u^{y}_n$, scale $\gamma_{u^y}$};
  \end{scope}

\end{tikzpicture}
