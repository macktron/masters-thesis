% =====================================================================
% Geometry of RoPE on one 2D plane of an attention head.
% Left: projected query q is rotated by R(p) through a modest absolute
% phase gamma*p*theta_m (\cref{eq:method:rope_block}).
% Right: after both query and key are rotated, the enclosed angle
% (and hence the inner product) depends only on the coordinate
% difference (\cref{eq:method:rope}).
% Loaded from sections/04_method/04_attention_variants.tex via \input.
% =====================================================================
\begin{tikzpicture}[
    font=\small,
    >=Latex,
    axis/.style={->, gray!55, thin},
    guide/.style={gray!28, thin},
    qraw/.style={->, thick, densely dashed, color=black!45},
    qvec/.style={->, very thick, color=blue!70!black},
    kraw/.style={->, thick, densely dashed, color=red!40!black},
    kvec/.style={->, very thick, color=red!70!black},
    angarc/.style={->, thin, color=black!65},
  ]

  % =================== LEFT PANEL: apply R(p) ===========================
  \begin{scope}[xshift=0mm]
    \def\Rad{18mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0.5*\Rad, 23mm)
      {Apply $\mat{R}(p)$ to the projected query};

    \draw[axis] (-20mm, 0) -- (22mm, 0);
    \draw[axis] (0, -8mm) -- (0, 21mm);
    \draw[guide] (0, 0) circle (\Rad);

    % gentle fan: content at 28°, then +10° and +18° of phase only
    \def\qbase{28}
    \def\aone{38}
    \def\atwo{46}

    \draw[qraw] (0, 0) -- (\qbase:\Rad);
    \node[color=black!50, font=\scriptsize, anchor=north west]
      at (\qbase:\Rad) {$\vect{q}=\mat{W}_q\vect{h}$};

    \draw[qvec, color=blue!50!black] (0, 0) -- (\aone:\Rad);
    \draw[qvec] (0, 0) -- (\atwo:\Rad);

    \node[color=blue!50!black, font=\scriptsize, anchor=west]
      at (\aone:\Rad) {$\mat{R}(p_i)\,\vect{q}$};
    \node[color=blue!70!black, font=\scriptsize, anchor=south west]
      at (\atwo:\Rad) {$\mat{R}(p_j)\,\vect{q}$};

    % small absolute-phase arc (first step only)
    \draw[angarc] (\qbase:9mm) arc (\qbase:\aone:9mm);
    \node[font=\scriptsize, anchor=west]
      at ({(\qbase+\aone)/2}:10.5mm) {$\gamma p_i\theta_m$};

    % tiny tick for the second step
    \draw[angarc, color=black!40] (\aone:7mm) arc (\aone:\atwo:7mm);

    \node[font=\scriptsize, anchor=north, align=center, text=black!60]
      at (0, -9mm)
      {same content $\vect{q}$; different physical $p$\\[-1pt]
       $\Rightarrow$ different absolute phase};
  \end{scope}

  % =================== RIGHT PANEL: relative phase ======================
  \begin{scope}[xshift=70mm]
    \def\Rad{18mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0.5*\Rad, 23mm)
      {Inner product sees only $p_j-p_i$};

    \draw[axis] (-20mm, 0) -- (22mm, 0);
    \draw[axis] (0, -8mm) -- (0, 21mm);
    \draw[guide] (0, 0) circle (\Rad);

    % content (unrotated) pair: angle φ = 16°
    \def\qrawang{22}
    \def\krawang{38}
    % after RoPE: both moved, relative extra phase ≈ 12°
    \def\qang{30}
    \def\kang{58}

    \draw[qraw] (0, 0) -- (\qrawang:\Rad);
    \draw[kraw] (0, 0) -- (\krawang:\Rad);
    \node[color=black!45, font=\scriptsize, anchor=north east]
      at (\qrawang:0.98*\Rad) {$\vect{q}_i$};
    \node[color=red!40!black, font=\scriptsize, anchor=south east]
      at (\krawang:0.98*\Rad) {$\vect{k}_j$};

    \draw[angarc, color=black!40] (\qrawang:6mm) arc (\qrawang:\krawang:6mm);
    \node[font=\scriptsize, color=black!50, anchor=south]
      at ({(\qrawang+\krawang)/2}:7.2mm) {$\varphi$};

    \draw[qvec] (0, 0) -- (\qang:\Rad);
    \draw[kvec] (0, 0) -- (\kang:\Rad);
    \node[color=blue!70!black, font=\scriptsize, anchor=west]
      at (\qang:\Rad) {$\tilde{\vect{q}}_i$};
    \node[color=red!70!black, font=\scriptsize, anchor=south west]
      at (\kang:\Rad) {$\tilde{\vect{k}}_j$};

    \draw[angarc] (\qang:11.5mm) arc (\qang:\kang:11.5mm);
    \node[font=\scriptsize, anchor=south west]
      at ({(\qang+\kang)/2}:12.8mm)
      {$\varphi{+}\gamma(p_j{-}p_i)\theta_m$};

    \node[font=\scriptsize, anchor=north, align=center, text=black!60]
      at (0, -9mm)
      {$\langle\tilde{\vect{q}}_i,\tilde{\vect{k}}_j\rangle
        =\vect{q}_i^{\top}\mat{R}(p_j{-}p_i)\,\vect{k}_j$};
  \end{scope}

\end{tikzpicture}
