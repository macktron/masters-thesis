% =====================================================================
% Geometry of RoPE on one 2D plane of an attention head.
% Left: absolute phase applied to the projected query.
% Right: content angle phi (dashed) vs. RoPE-modified enclosed angle;
% the inner product depends only on the coordinate difference.
% Labels sit outside the unit circle to avoid overlap.
% Loaded from sections/04_method/04_attention_variants.tex via \input.
% =====================================================================
\begin{tikzpicture}[
    font=\small,
    >=Latex,
    axis/.style={->, gray!55, thin},
    guide/.style={gray!30, thin},
    qraw/.style={->, thick, densely dashed, color=black!50},
    qvec/.style={->, very thick, color=blue!70!black},
    kraw/.style={->, thick, densely dashed, color=red!45!black},
    kvec/.style={->, very thick, color=red!70!black},
    angarc/.style={->, thin, color=black!65},
  ]

  % =================== LEFT PANEL =======================================
  \begin{scope}[xshift=0mm]
    \def\Rad{18mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (3mm, 26mm)
      {Apply $\mat{R}(p)$ to the projected query};

    \draw[axis] (-21mm, 0) -- (24mm, 0);
    \draw[axis] (0, -10mm) -- (0, 24mm);
    \draw[guide] (0, 0) circle (\Rad);

    % clearly visible rotation (~28°)
    \def\qbase{22}
    \def\arot{50}

    \draw[qraw] (0, 0) -- (\qbase:\Rad);
    \node[color=black!55, font=\scriptsize, anchor=north west]
      at (\qbase:1.18*\Rad) {$\vect{q}=\mat{W}_q\vect{h}$};

    \draw[qvec] (0, 0) -- (\arot:\Rad);
    \node[color=blue!70!black, font=\scriptsize, anchor=south west]
      at (\arot:1.18*\Rad) {$\tilde{\vect{q}}=\mat{R}(p)\,\vect{q}$};

    % arc inside the circle; label outside with a short leader
    \draw[angarc] (\qbase:9mm) arc (\qbase:\arot:9mm);
    \draw[gray!55, thin]
      ({(\qbase+\arot)/2}:9mm) -- ({(\qbase+\arot)/2}:1.22*\Rad);
    \node[font=\scriptsize, anchor=west]
      at ({(\qbase+\arot)/2}:1.25*\Rad) {$\gamma p\theta_m$};

    \node[font=\scriptsize, anchor=north, align=center, text=black!60]
      at (2mm, -11mm)
      {projection first, then a planar rotation\\[-1pt]
       by the physical coordinate $p$};
  \end{scope}

  % =================== RIGHT PANEL ======================================
  \begin{scope}[xshift=72mm]
    \def\Rad{18mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (3mm, 26mm)
      {Inner product sees only $p_j-p_i$};

    \draw[axis] (-21mm, 0) -- (24mm, 0);
    \draw[axis] (0, -10mm) -- (0, 24mm);
    \draw[guide] (0, 0) circle (\Rad);

    % content angle φ ≈ 18°; relative phase ≈ 22° (readable gap)
    \def\qrawang{18}
    \def\krawang{36}
    \def\qang{32}
    \def\kang{72}

    \draw[qraw] (0, 0) -- (\qrawang:\Rad);
    \draw[kraw] (0, 0) -- (\krawang:\Rad);
    \node[color=black!50, font=\scriptsize, anchor=north east]
      at (\qrawang:1.16*\Rad) {$\vect{q}_i$};
    \node[color=red!45!black, font=\scriptsize, anchor=west]
      at (\krawang:1.16*\Rad) {$\vect{k}_j$};

    % small content-angle arc + label inside, near the origin
    \draw[angarc, color=black!45] (\qrawang:6mm) arc (\qrawang:\krawang:6mm);
    \node[font=\scriptsize, color=black!55, anchor=west]
      at ({(\qrawang+\krawang)/2}:7.2mm) {$\varphi$};

    \draw[qvec] (0, 0) -- (\qang:\Rad);
    \draw[kvec] (0, 0) -- (\kang:\Rad);
    \node[color=blue!70!black, font=\scriptsize, anchor=south east]
      at (\qang:1.16*\Rad) {$\tilde{\vect{q}}_i$};
    \node[color=red!70!black, font=\scriptsize, anchor=south west]
      at (\kang:1.18*\Rad) {$\tilde{\vect{k}}_j$};

    % relative-phase arc; long label kept fully outside the circle
    \draw[angarc] (\qang:11mm) arc (\qang:\kang:11mm);
    \coordinate (relmid) at ({(\qang+\kang)/2}:11mm);
    \coordinate (rellbl) at (1.55*\Rad, 1.05*\Rad);
    \draw[gray!55, thin] (relmid) -- (rellbl);
    \node[font=\scriptsize, anchor=west, align=left] at (rellbl)
      {$\varphi+\gamma(p_j-p_i)\theta_m$};

    \node[font=\scriptsize, anchor=north, align=center, text=black!60]
      at (2mm, -11mm)
      {$\langle\tilde{\vect{q}}_i,\tilde{\vect{k}}_j\rangle
        =\vect{q}_i^{\top}\mat{R}(p_j-p_i)\,\vect{k}_j$};
  \end{scope}

\end{tikzpicture}
