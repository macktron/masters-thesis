% =====================================================================
% Geometry of rotary positional encoding (\gls{rope}) on a single
% two-dimensional plane of an attention head.  Left panel: the
% projected query q (dashed) is rotated by the window-normalised TOA
% of its pulse at the fixed rate gamma*theta_m, so irregularly spaced
% arrival times t_1 < t_2 < t_3 map to correspondingly irregular
% absolute phases (\cref{eq:method:rope_block}).  Right panel: the
% angle enclosed between a rotated query and a rotated key is the
% content angle plus gamma*(p_j-p_i)*theta_m, so the inner product---
% and hence the attention logit---depends only on the coordinate
% difference (\cref{eq:method:rope}).  Loaded from
% sections/04_method/04_attention_variants.tex via \input{...}.
% =====================================================================
\begin{tikzpicture}[
    font=\small,
    >=Latex,
    axis/.style={->, gray!60, thin},
    guide/.style={gray!35, thin},
    qvec/.style={->, thick, color=blue!70!black},
    kvec/.style={->, thick, color=red!70!black},
    angarc/.style={->, thin, color=black!75},
  ]

  % =================== LEFT PANEL: rotation by coordinate ===============
  \begin{scope}[xshift=0mm]
    \def\Rad{16mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0, 21mm)
      {Rotation by coordinate};

    % axes and reference circle
    \draw[axis] (-19mm, 0) -- (19mm, 0);
    \draw[axis] (0, -6mm) -- (0, 20mm);
    \draw[guide] (0, 0) circle (\Rad);

    \def\qbase{10}   % content angle of the projected query
    \def\rate{40}    % gamma*theta_m in degrees per unit of coordinate

    % unrotated projected query
    \draw[qvec, dashed, color=black!55] (0, 0) -- (\qbase:\Rad);
    \node[color=black!55, font=\scriptsize, anchor=west]
      at (\qbase:\Rad) {$\vect{q}$};

    % rotations by irregularly spaced window-normalised TOA values
    % t_1 = 0.20, t_2 = 0.45, t_3 = 0.90
    \foreach \t/\sh in {0.20/45, 0.45/70, 0.90/95} {
      \pgfmathsetmacro{\ang}{\qbase + \t*\rate}
      \draw[qvec, color=blue!\sh!black] (0, 0) -- (\ang:\Rad);
    }
    \pgfmathsetmacro{\aone}{\qbase + 0.20*\rate}
    \node[color=blue!45!black, font=\scriptsize, anchor=west]
      at (\aone:\Rad) {$\mat{R}(t_1)\,\vect{q}$};
    \pgfmathsetmacro{\athree}{\qbase + 0.90*\rate}
    \node[color=blue!95!black, font=\scriptsize, anchor=south west]
      at (\athree:\Rad) {$\mat{R}(t_3)\,\vect{q}$};

    % absolute phase of the first rotation
    \draw[angarc] (\qbase:9mm) arc (\qbase:\aone:9mm);
    \pgfmathsetmacro{\amid}{(\qbase + \aone)/2}
    \node[font=\scriptsize, anchor=west] at (\amid:10.5mm)
      {$\gamma t_1\theta_m$};
  \end{scope}

  % =================== RIGHT PANEL: relative phase ===================
  \begin{scope}[xshift=62mm]
    \def\Rad{16mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0, 21mm)
      {Relative phase};

    % axes and reference circle
    \draw[axis] (-19mm, 0) -- (19mm, 0);
    \draw[axis] (0, -6mm) -- (0, 20mm);
    \draw[guide] (0, 0) circle (\Rad);

    \def\qang{16}
    \def\kang{68}

    % rotated query and key
    \draw[qvec] (0, 0) -- (\qang:\Rad);
    \draw[kvec] (0, 0) -- (\kang:\Rad);

    \node[color=blue!70!black, font=\scriptsize, anchor=west]
      at (\qang:\Rad) {$\tilde{\vect{q}}_i$};
    \node[color=red!70!black, font=\scriptsize, anchor=south east]
      at (\kang:\Rad) {$\tilde{\vect{k}}_j$};

    % enclosed angle arc
    \draw[angarc] (\qang:10mm) arc (\qang:\kang:10mm);
    \pgfmathsetmacro{\mid}{(\qang + \kang)/2}
    \node[font=\scriptsize, anchor=south west] at (\mid:11mm)
      {$\varphi + \gamma(p_j{-}p_i)\theta_m$};
  \end{scope}

\end{tikzpicture}
