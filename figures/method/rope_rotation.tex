% =====================================================================
% Geometry of rotary positional encoding (\gls{rope}) on a single
% two-dimensional plane of an attention head.  Left panel: a query
% vector is rotated at a fixed angular rate $\theta_m$ as its coordinate
% $p$ increases, so the absolute coordinate maps to an absolute phase
% (\cref{eq:method:rope_block}).  Right panel: the angle enclosed
% between a rotated query and a rotated key is the content angle plus
% $(p_j-p_i)\theta_m$, so the inner product---and hence the attention
% logit---depends only on the coordinate difference $p_j-p_i$
% (\cref{eq:method:rope}).  Loaded from
% sections/04_method/04_attention_variants.tex via \input{...}.
% =====================================================================
\begin{tikzpicture}[
    font=\small,
    >=Latex,
    axis/.style={->, gray!60, thin},
    guide/.style={gray!35, thin},
    qvec/.style={->, thick, color=blue!70!black},
    kvec/.style={->, thick, color=red!70!black},
    angarc/.style={->, thin, color=black!75},
  ]

  % =================== LEFT PANEL: absolute rotation ===================
  \begin{scope}[xshift=0mm]
    \def\Rad{16mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0, 21mm)
      {Rotation by position};

    % axes and reference circle
    \draw[axis] (-19mm, 0) -- (19mm, 0);
    \draw[axis] (0, -6mm) -- (0, 20mm);
    \draw[guide] (0, 0) circle (\Rad);

    % fan of rotated query vectors: base angle 8deg, increment 20deg
    \def\base{8}
    \def\inc{20}
    \foreach \i/\sh in {0/30, 1/50, 2/70, 3/90} {
      \pgfmathsetmacro{\ang}{\base + \i*\inc}
      \draw[qvec, color=blue!\sh!black] (0, 0) -- (\ang:\Rad);
    }

    % labels for first and last vector
    \node[color=blue!30!black, font=\scriptsize, anchor=west]
      at (\base:\Rad) {$\vect{q}=\mat{R}(0)\,\vect{q}$};
    \pgfmathsetmacro{\last}{\base + 3*\inc}
    \node[color=blue!90!black, font=\scriptsize, anchor=south west]
      at (\last:\Rad) {$\mat{R}(3)\,\vect{q}$};

    % angular increment arc between i=1 and i=2
    \pgfmathsetmacro{\aone}{\base + \inc}
    \pgfmathsetmacro{\atwo}{\base + 2*\inc}
    \draw[angarc] (\aone:9mm) arc (\aone:\atwo:9mm);
    \pgfmathsetmacro{\amid}{\base + 1.5*\inc}
    \node[font=\scriptsize] at (\amid:11.5mm) {$\theta_m$};
  \end{scope}

  % =================== RIGHT PANEL: relative phase ===================
  \begin{scope}[xshift=62mm]
    \def\Rad{16mm}
    \node[font=\footnotesize\bfseries, anchor=south] at (0, 21mm)
      {Relative phase};

    % axes and reference circle
    \draw[axis] (-19mm, 0) -- (19mm, 0);
    \draw[axis] (0, -6mm) -- (0, 20mm);
    \draw[guide] (0, 0) circle (\Rad);

    \def\qang{18}
    \def\kang{96}

    % rotated query and key
    \draw[qvec] (0, 0) -- (\qang:\Rad);
    \draw[kvec] (0, 0) -- (\kang:\Rad);

    \node[color=blue!70!black, font=\scriptsize, anchor=west]
      at (\qang:\Rad) {$\tilde{\vect{q}}_i$};
    \node[color=red!70!black, font=\scriptsize, anchor=south east]
      at (\kang:\Rad) {$\tilde{\vect{k}}_j$};

    % enclosed angle arc
    \draw[angarc] (\qang:10mm) arc (\qang:\kang:10mm);
    \pgfmathsetmacro{\mid}{(\qang + \kang)/2}
    \node[font=\scriptsize, anchor=south west] at (\mid:11mm)
      {$\varphi + (p_j{-}p_i)\theta_m$};
  \end{scope}

\end{tikzpicture}
